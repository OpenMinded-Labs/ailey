💻 Hardware-Anforderungen

🎮 GPU (Empfohlen)

  • Minimum: RTX 3060 (12GB VRAM)
  • Optimal: RTX 4080/4090 (16-24GB VRAM)
  • Profi: A100/H100 (40-80GB VRAM)
  • Budget: RTX 3080 (10GB VRAM)

🧠 RAM

  • Minimum: 16GB DDR4
  • Empfohlen: 32GB DDR4/DDR5
  • Optimal: 64GB+ für große Datensätze
  • Typ: DDR4-3200 oder DDR5-4800

💾 Speicher

  • System: 1TB NVMe SSD
  • Modelle: 500GB+ zusätzlich
  • Datensätze: 2TB+ HDD/SSD
  • Geschwindigkeit: PCIe 4.0 NVMe

⚡ CPU

  • Intel: i7-12700K oder besser
  • AMD: Ryzen 7 5800X oder besser
  • Kerne: 8+ Kerne, 16+ Threads
  • Takt: 3.5GHz+ Base Clock
⚠️ Wichtiger Hinweis: LoRA-Training kann auch auf schwächerer Hardware durchgeführt werden, dauert aber entsprechend länger. Für Experimente reicht auch eine RTX 3060 mit 12GB VRAM.

🚀 Ollama Installation & Setup

1. Ollama installieren

# Linux/macOS curl -fsSL https://ollama.ai/install.sh | sh # Windows (PowerShell als Administrator) winget install ollama # Docker Alternative docker run -d -v ollama:/root/.ollama -p 11434:11434 --name ollama ollama/ollama

# Umgebungsvariablen setzen export OLLAMA_HOST=0.0.0.0:11434 export OLLAMA_MODELS=/path/to/models export OLLAMA_NUM_PARALLEL=4 export OLLAMA_MAX_LOADED_MODELS=3 # GPU-Speicher begrenzen (optional) export OLLAMA_GPU_MEMORY_FRACTION=0.8

# train_lora.py import torch from transformers import AutoTokenizer, AutoModelForCausalLM from peft import LoraConfig, get_peft_model, TaskType from datasets import load_dataset # Modell und Tokenizer laden model_name = "mistralai/Mistral-7B-v0.1" tokenizer = AutoTokenizer.from_pretrained(model_name) model = AutoModelForCausalLM.from_pretrained( model_name, torch_dtype=torch.float16, device_map="auto", load_in_4bit=True ) # LoRA-Konfiguration lora_config = LoraConfig( task_type=TaskType.CAUSAL_LM, r=16, # Rank lora_alpha=32, lora_dropout=0.1, target_modules=["q_proj", "v_proj", "k_proj", "o_proj"] ) # LoRA-Modell erstellen model = get_peft_model(model, lora_config) # Training-Parameter training_args = { "output_dir": "./lora-mistral-7b", "num_train_epochs": 3, "per_device_train_batch_size": 4, "gradient_accumulation_steps": 4, "learning_rate": 2e-4, "warmup_steps": 100, "logging_steps": 10, "save_steps": 500, "evaluation_strategy": "steps", "eval_steps": 500 }

4. Training starten

# Training ausführen python train_lora.py \ --model_name mistralai/Mistral-7B-v0.1 \ --dataset_path ./processed_data \ --output_dir ./lora-output \ --num_epochs 3 \ --batch_size 4 \ --learning_rate 2e-4 # Training mit Weights & Biases überwachen wandb login python train_lora.py --use_wandb

📚 RAG System Implementation

1. Vector Database Setup

# ChromaDB installieren pip install chromadb sentence-transformers # Qdrant Alternative pip install qdrant-client # Pinecone Alternative pip install pinecone-client

2. RAG-Pipeline erstellen

# rag_system.py import chromadb from sentence_transformers import SentenceTransformer import requests import json class RAGSystem: def __init__(self): # Vector Database initialisieren self.chroma_client = chromadb.Client() self.collection = self.chroma_client.create_collection("knowledge_base") # Embedding-Modell laden self.embedding_model = SentenceTransformer('all-MiniLM-L6-v2') # Ollama-Verbindung self.ollama_url = "http://localhost:11434/api/generate" def add_documents(self, documents, metadata=None): """Dokumente zur Wissensbasis hinzufügen""" embeddings = self.embedding_model.encode(documents) self.collection.add( embeddings=embeddings.tolist(), documents=documents, metadatas=metadata or [{}] * len(documents), ids=[f"doc_{i}" for i in range(len(documents))] ) def retrieve_context(self, query, n_results=3): """Relevante Kontexte abrufen""" query_embedding = self.embedding_model.encode([query]) results = self.collection.query( query_embeddings=query_embedding.tolist(), n_results=n_results ) return results['documents'][0] def generate_response(self, query, context): """Antwort mit Kontext generieren""" prompt = f""" Kontext: {' '.join(context)} Frage: {query} Antwort basierend auf dem Kontext: """ response = requests.post(self.ollama_url, json={ "model": "mistral:7b", "prompt": prompt, "stream": False }) return response.json()['response'] def query(self, question): """Vollständige RAG-Pipeline""" context = self.retrieve_context(question) response = self.generate_response(question, context) return response, context # RAG-System verwenden rag = RAGSystem() # Dokumente hinzufügen documents = [ "Python ist eine interpretierte Programmiersprache.", "Machine Learning verwendet Algorithmen zur Mustererkennung.", "Neuronale Netze sind von biologischen Neuronen inspiriert." ] rag.add_documents(documents) # Frage stellen answer, context = rag.query("Was ist Python?") print(f"Antwort: {answer}") print(f"Kontext: {context}")

3. Erweiterte RAG-Features

# Hybrid Search (Keyword + Semantic) from rank_bm25 import BM25Okapi import numpy as np class HybridRAG(RAGSystem): def __init__(self): super().__init__() self.bm25 = None self.documents = [] def add_documents(self, documents, metadata=None): super().add_documents(documents, metadata) # BM25 für Keyword-Search tokenized_docs = [doc.lower().split() for doc in documents] self.bm25 = BM25Okapi(tokenized_docs) self.documents.extend(documents) def hybrid_retrieve(self, query, n_results=3, alpha=0.7): """Hybrid Retrieval: Semantic + Keyword""" # Semantic Search semantic_results = self.retrieve_context(query, n_results*2) # Keyword Search tokenized_query = query.lower().split() bm25_scores = self.bm25.get_scores(tokenized_query) # Scores kombinieren # Implementation der Score-Kombination... return semantic_results[:n_results] # Document Chunking für große Texte def chunk_document(text, chunk_size=500, overlap=50): """Text in überlappende Chunks aufteilen""" chunks = [] start = 0 while start < len(text): end = start + chunk_size chunk = text[start:end] chunks.append(chunk) start = end - overlap return chunks

// server.js const express = require('express'); const axios = require('axios'); const cors = require('cors'); const app = express(); app.use(cors()); app.use(express.json()); // Ollama-API Wrapper app.post('/api/chat', async (req, res) => { try { const { message, model = 'mistral:7b', history = [] } = req.body; // Kontext aus Historie erstellen const context = history.map(msg => `${msg.role === 'user' ? 'Human' : 'Assistant'}: ${msg.content}` ).join('\n'); const prompt = context ? `${context}\nHuman: ${message}\nAssistant:` : message; const response = await axios.post('http://localhost:11434/api/generate', { model: model, prompt: prompt, stream: false, options: { temperature: 0.7, top_p: 0.9, max_tokens: 1000 } }); res.json({ response: response.data.response }); } catch (error) { console.error('Ollama API Fehler:', error); res.status(500).json({ error: 'Fehler bei der KI-Anfrage' }); } }); // RAG-Endpoint app.post('/api/rag-query', async (req, res) => { try { const { query } = req.body; // RAG-System aufrufen (Python-Service) const ragResponse = await axios.post('http://localhost:5000/rag', { query: query }); res.json(ragResponse.data); } catch (error) { res.status(500).json({ error: 'RAG-System Fehler' }); } }); app.listen(3001, () => { console.log('Backend läuft auf Port 3001'); });

2. Python-Client

# ollama_client.py import requests import json import sseclient # pip install sseclient-py class OllamaClient: def __init__(self, base_url="http://localhost:11434"): self.base_url = base_url def generate(self, model, prompt, stream=False, **options): """Text generieren""" data = { "model": model, "prompt": prompt, "stream": stream, "options": options } response = requests.post(f"{self.base_url}/api/generate", json=data, stream=stream) if stream: return self._handle_stream(response) else: return response.json()['response'] def chat(self, model, messages, stream=False): """Chat mit Konversationshistorie""" data = { "model": model, "messages": messages, "stream": stream } response = requests.post(f"{self.base_url}/api/chat", json=data, stream=stream) if stream: return self._handle_stream(response) else: return response.json()['message']['content'] def _handle_stream(self, response): """Streaming-Response verarbeiten""" client = sseclient.SSEClient(response) for event in client.events(): if event.data: data = json.loads(event.data) if 'response' in data: yield data['response'] elif 'message' in data: yield data['message']['content'] # Verwendung client = OllamaClient() # Einfache Generation response = client.generate("mistral:7b", "Was ist Machine Learning?") print(response) # Streaming for token in client.generate("mistral:7b", "Erzähle eine Geschichte", stream=True): print(token, end='', flush=True) # Chat messages = [ {"role": "user", "content": "Hallo!"}, {"role": "assistant", "content": "Hallo! Wie kann ich helfen?"}, {"role": "user", "content": "Erkläre mir Python"} ] response = client.chat("mistral:7b", messages) print(response)

# Hugging Face Integration from transformers import pipeline import requests class HybridAISystem: def __init__(self): self.ollama_client = OllamaClient() # Hugging Face für spezielle Tasks self.sentiment_analyzer = pipeline("sentiment-analysis") self.summarizer = pipeline("summarization") def analyze_and_respond(self, text): """Text analysieren und kontextuell antworten""" # Sentiment-Analyse sentiment = self.sentiment_analyzer(text)[0] # Zusammenfassung bei langen Texten if len(text) > 1000: summary = self.summarizer(text, max_length=150)[0]['summary_text'] context = f"Zusammenfassung: {summary}\nSentiment: {sentiment['label']}" else: context = f"Text: {text}\nSentiment: {sentiment['label']}" # Ollama für Antwort prompt = f""" Kontext: {context} Bitte antworte angemessen auf den Text unter Berücksichtigung des Sentiments. """ return self.ollama_client.generate("mistral:7b", prompt) # OpenAI API als Fallback import openai class FallbackSystem: def __init__(self, openai_key): self.ollama_client = OllamaClient() openai.api_key = openai_key def generate_with_fallback(self, prompt, model="mistral:7b"): """Ollama mit OpenAI-Fallback""" try { # Zuerst Ollama versuchen return self.ollama_client.generate(model, prompt) } catch (Exception e) { print(f"Ollama-Fehler: {e}, verwende OpenAI-Fallback") # OpenAI als Fallback response = openai.ChatCompletion.create( model="gpt-3.5-turbo", messages=[{"role": "user", "content": prompt}] ) return response.choices[0].message.content }
